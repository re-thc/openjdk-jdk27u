Baseline adapter source: ceca9846ff55b499df85996a3dfce7585feb0e89
After adapter: working tree review fixes. Both used the same Cargo.lock,

The pre-fix crate was staged outside the repository and built with:
cargo build --frozen --release --target x86_64-unknown-linux-gnu --target-dir /tmp/regex-review-before-target
Each static library was linked into a separate DSO:
gcc -shared -o /tmp/regex-review-<revision>.so -Wl,--whole-archive <staticlib> -Wl,--no-whole-archive -Wl,--version-script=/tmp/regex-review-exports.map -ldl -lpthread -lm
Export map: { global: jdk_regex_compile; jdk_regex_free; jdk_regex_may_match; local: *; };
Only the three C ABI symbols were exported; other symbols were local.
The caller was compiled with g++ -O3 -ldl, pinned to CPU 0, with competing
compilation paused. Each size used 20,000 warmup calls and 12 pairs of
200,000-call batches, alternating revision order within successive pairs.
This measures only the native boundary/search, not Matcher.find.

Caller source:
#include <chrono>
#include <cstdint>
#include <cstdio>
#include <cstdlib>
#include <dlfcn.h>
#include <string>
#include <vector>
using Compile = void* (*)(const unsigned char*, size_t);
using Search = unsigned char (*)(const void*, const unsigned char*, size_t);
using Free = void (*)(void*);
struct Adapter { void* library; void* handle; Search search; Free free; };
int main() {
    std::vector<Adapter> adapters;
    for (const char* path : {"/tmp/regex-review-before.so", "/tmp/regex-review-after.so"}) {
        void* lib = dlopen(path, RTLD_NOW | RTLD_LOCAL);
        if (!lib) { std::fprintf(stderr, "%s\n", dlerror()); return 1; }
        auto compile = reinterpret_cast<Compile>(dlsym(lib,"jdk_regex_compile"));
        constexpr char pattern[] = "error[0-9]+";
        Adapter a{lib, compile(reinterpret_cast<const unsigned char*>(pattern),sizeof(pattern)-1),
                reinterpret_cast<Search>(dlsym(lib,"jdk_regex_may_match")),
                reinterpret_cast<Free>(dlsym(lib,"jdk_regex_free"))};
        if (!a.handle) return 2;
        adapters.push_back(a);
    }
    std::puts("bytes,pair,revision,ns_per_call");
    for (size_t length : {4096u,32768u}) {
        std::string text(length,'x');
        for (size_t i=1; i<length; i+=2) text[i]=' ';
        for (auto& a : adapters)
            for (int i=0; i<20000; i++)
                if(a.search(a.handle,reinterpret_cast<const unsigned char*>(text.data()),length)) return 3;
        for (int pair=0; pair<12; pair++) {
            for (int order=0; order<2; order++) {
                int index=(order+pair)%2; auto& a=adapters[index];
                auto start=std::chrono::steady_clock::now();
                for (int i=0; i<200000; i++)
                    if(a.search(a.handle,reinterpret_cast<const unsigned char*>(text.data()),length)) return 4;
                double elapsed=std::chrono::duration<double,std::nano>(std::chrono::steady_clock::now()-start).count()/200000;
                std::printf("%zu,%d,%s,%.3f\n",length,pair,index ? "after" : "before",elapsed);
            }
        }
    }
    for (auto& a : adapters) { a.free(a.handle); dlclose(a.library); }
}
