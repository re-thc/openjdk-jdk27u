public class ThresholdProbe {
    static long mask64(int bits) { return bits >= 64 ? -1L : (1L << bits) - 1; }
    static long direct(int bits, long v) { if (bits > 63) return Long.MIN_VALUE; return v & mask64(bits); }
    static long normalized(int bits, long v) { if (bits >= 64) return Long.MIN_VALUE; return v & mask64(bits); }
    public static void main(String[] args) {
        long sum = 0;
        for (int i = 0; i < 100000; i++) {
            int b = (i % 130) - 65;
            sum += direct(b, i); sum += normalized(b, i);
        }
        System.out.println(sum);
    }
}
